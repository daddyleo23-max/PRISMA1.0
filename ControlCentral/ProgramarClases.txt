from PyQt6.QtWidgets import (
    QWidget, QVBoxLayout, QLabel, QComboBox, QLineEdit, QPushButton,
    QHBoxLayout, QFormLayout, QCheckBox, QTimeEdit, QMessageBox
)
from PyQt6.QtCore import Qt, QTime
import json
import os

class ProgramarClases(QWidget):
    def __init__(self, parent=None):
        super().__init__(parent)
        self.setStyleSheet("""
            QWidget {
                background-color: #23272f;
                border-radius: 18px;
            }
            QLabel#titulo {
                color: #00e1ff;
                font-size: 28px;
                font-weight: bold;
                margin-bottom: 18px;
            }
            QLabel {
                color: #A3BE8C;
                font-size: 20px;
                font-weight: bold;
            }
            QLineEdit, QComboBox {
                background: #2E3440;
                color: #00e1ff;
                font-size: 20px;
                border: 2px solid #00e1ff;
                border-radius: 10px;
                padding: 10px;
                margin-bottom: 12px;
            }
            QCheckBox {
                color: #A3BE8C;
                font-size: 18px;
            }
            QTimeEdit {
                background: #2E3440;
                color: #00e1ff;
                font-size: 20px;
                border: 2px solid #00e1ff;
                border-radius: 10px;
                padding: 8px;
                margin-bottom: 12px;
            }
            QPushButton {
                background-color: #00e1ff;
                color: #1e2229;
                font-size: 22px;
                border-radius: 16px;
                font-weight: bold;
                padding: 14px 0;
                min-width: 180px;
            }
            QPushButton:hover {
                background-color: #00b3cc;
            }
        """)

        layout = QVBoxLayout(self)
        layout.setContentsMargins(60, 40, 60, 40)
        layout.setSpacing(24)

        lbl_titulo = QLabel("Programar Clase")
        lbl_titulo.setObjectName("titulo")
        lbl_titulo.setAlignment(Qt.AlignmentFlag.AlignCenter)
        layout.addWidget(lbl_titulo)

        form = QFormLayout()
        self.combo_clase = QComboBox()
        self.cargar_clases()
        form.addRow("Clase:", self.combo_clase)

        self.input_fecha_inicio = QLineEdit()
        self.input_fecha_inicio.setPlaceholderText("Ejemplo: 2025-01-31")
        form.addRow("Fecha de Inicio:", self.input_fecha_inicio)

        self.input_fecha_fin = QLineEdit()
        self.input_fecha_fin.setPlaceholderText("Ejemplo: 2025-06-17")
        form.addRow("Fecha de Fin:", self.input_fecha_fin)

        # Días de la semana
        dias = ["Lunes", "Martes", "Miércoles", "Jueves", "Viernes", "Sábado", "Domingo"]
        self.checkboxes_dias = []
        dias_layout = QHBoxLayout()
        for dia in dias:
            cb = QCheckBox(dia)
            dias_layout.addWidget(cb)
            self.checkboxes_dias.append(cb)
        form.addRow("Días:", dias_layout)

        # Hora de inicio y fin
        horas_layout = QHBoxLayout()
        self.time_inicio = QTimeEdit()
        self.time_inicio.setTime(QTime(8, 0))
        self.time_fin = QTimeEdit()
        self.time_fin.setTime(QTime(10, 0))
        horas_layout.addWidget(QLabel("Inicio:"))
        horas_layout.addWidget(self.time_inicio)
        horas_layout.addWidget(QLabel("Fin:"))
        horas_layout.addWidget(self.time_fin)
        form.addRow("Horario:", horas_layout)

        layout.addLayout(form)

        btn_guardar = QPushButton("Guardar Programación")
        btn_guardar.clicked.connect(self.guardar_programacion)
        layout.addWidget(btn_guardar)

        layout.addStretch()

    def cargar_clases(self):
        self.combo_clase.clear()
        try:
            with open("clases.json", "r", encoding="utf-8") as file:
                clases = json.load(file)
                for clase in clases:
                    self.combo_clase.addItem(f"{clase['materia']} - {clase['grupo']}")
        except Exception:
            pass

    def guardar_programacion(self):
        clase = self.combo_clase.currentText()
        fecha_inicio = self.input_fecha_inicio.text().strip()
        fecha_fin = self.input_fecha_fin.text().strip()
        dias = [cb.text() for cb in self.checkboxes_dias if cb.isChecked()]
        hora_inicio = self.time_inicio.time().toString("HH:mm")
        hora_fin = self.time_fin.time().toString("HH:mm")

        if not clase or not fecha_inicio or not fecha_fin:
            QMessageBox.warning(self, "Error", "Todos los campos son obligatorios.")
            return
        if not dias:
            QMessageBox.warning(self, "Error", "Selecciona al menos un día.")
            return
        if hora_inicio >= hora_fin:
            QMessageBox.warning(self, "Error", "La hora de inicio debe ser antes que la de fin.")
            return

        # Guardar en programaciones.json
        try:
            with open("programaciones.json", "r", encoding="utf-8") as file:
                programaciones = json.load(file)
        except Exception:
            programaciones = []

        programaciones.append({
            "clase": clase,
            "fecha_inicio": fecha_inicio,
            "fecha_fin": fecha_fin,
            "dias": dias,
            "hora_inicio": hora_inicio,
            "hora_fin": hora_fin
        })

        with open("programaciones.json", "w", encoding="utf-8") as file:
            json.dump(programaciones, file, indent=4, ensure_ascii=False)

        QMessageBox.information(self, "Éxito", "Clase programada correctamente.")
        # Limpiar campos
        self.input_fecha_inicio.clear()
        self.input_fecha_fin.clear()
        for cb in self.checkboxes_dias:
            cb.setChecked(False)
        self.time_inicio.setTime(QTime(8, 0))
        self.time_fin.setTime(QTime(10, 0))
